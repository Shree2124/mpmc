
import math

class FuzzyMembership:

    def triangular(self, x, a, b, c):
        if x <= a or x >= c:
            return 0.0
        elif x == b:
            return 1.0
        elif x < b:
            return (x - a) / (b - a)
        else:
            return (c - x) / (c - b)

    def trapezoidal(self, x, a, b, c, d):
        if x <= a or x >= d:
            return 0.0
        elif b <= x <= c:
            return 1.0
        elif x < b:
            return (x - a) / (b - a)
        else:
            return (d - x) / (d - c)

    def gaussian(self, x, c, sigma):
        return math.exp(-((x - c) ** 2) / (2 * sigma ** 2))


# Main Program
print("FUZZY MEMBERSHIP FUNCTIONS")
print("--------------------------")

# User Input Values
n = int(input("Enter number of input values: "))

values = []

print("\nEnter input values:")

for i in range(n):
    value = float(input("Enter value " + str(i + 1) + ": "))
    values.append(value)


# Triangular Parameters
print("\nEnter Triangular Membership parameters")

a = float(input("Enter a: "))
b = float(input("Enter b: "))
c = float(input("Enter c: "))

triangular_parameters = (a, b, c)


# Trapezoidal Parameters
print("\nEnter Trapezoidal Membership parameters")

a = float(input("Enter a: "))
b = float(input("Enter b: "))
c = float(input("Enter c: "))
d = float(input("Enter d: "))

trapezoidal_parameters = (a, b, c, d)


# Gaussian Parameters
print("\nEnter Gaussian Membership parameters")

center = float(input("Enter center: "))
sigma = float(input("Enter sigma: "))

gaussian_parameters = (center, sigma)


# Create Object
fuzzy = FuzzyMembership()


# Display Results
print("\nFUZZY MEMBERSHIP RESULTS")
print("---------------------------------------------")
print("Input\tTriangular\tTrapezoidal\tGaussian")
print("---------------------------------------------")

for value in values:

    triangular = fuzzy.triangular(
        value, *triangular_parameters
    )

    trapezoidal = fuzzy.trapezoidal(
        value, *trapezoidal_parameters
    )

    gaussian = fuzzy.gaussian(
        value, *gaussian_parameters
    )

    print(
        f"{value:g}\t{triangular:.3f}\t\t"
        f"{trapezoidal:.3f}\t\t{gaussian:.3f}"
    )

print("---------------------------------------------")
