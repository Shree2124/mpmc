
import numpy as np

# User Input
print("Perceptron Learning Algorithm - AND Gate")

X = []
Y = []

print("\nEnter 4 input combinations (X1, X2) and target output:")

for i in range(4):
    x1 = int(input("Enter X1: "))
    x2 = int(input("Enter X2: "))
    y = int(input("Enter Target Output: "))

    X.append([x1, x2])
    Y.append(y)

X = np.array(X)
Y = np.array(Y)

# Initialize Weights and Bias
W = np.array([0.0, 0.0])
b = -0.5
lr = 0.1

# Activation Function
def activation(x):
    return 1 if x >= 0 else 0

# Training
for epoch in range(10):
    for i in range(len(X)):
        output = activation(np.dot(X[i], W) + b)

        error = Y[i] - output

        W += lr * error * X[i]
        b += lr * error

# Display Final Weights and Bias
print("\nFinal Weights:", W)
print("Final Bias:", b)

# Display Final Output
print("\nFinal Output:")

for x in X:
    output = activation(np.dot(x, W) + b)
    print("Input =", x, "Output =", output)
